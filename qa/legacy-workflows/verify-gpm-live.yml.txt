name: Verify published GPM helper

on:
  push:
    branches: [main]
    paths: ['.github/workflows/verify-gpm-live.yml']
  workflow_dispatch:

permissions:
  contents: read

jobs:
  verify:
    if: github.repository == 'Maksimas-win/bukiskis-parama'
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v6
        with:
          persist-credentials: false
      - name: Check live Pages HTML, scripts, styles and all language corpora
        run: |
          python3 - <<'PY'
          import concurrent.futures, hashlib, json, os, pathlib, time, urllib.request
          root = pathlib.Path('docs')
          base = 'https://maksimas-win.github.io/bukiskis-parama/'
          languages = ['ru', 'lt', 'en', 'pl', 'de', 'uk']
          names = ['index.html', 'assets/js/gpm-helper.js', 'assets/css/gpm-helper.css']
          names += [f'assets/js/gpm-data-{lang}.js' for lang in languages]
          names += [f'{lang}/index.html' for lang in languages]
          names += [f'{lang}/privacy.html' for lang in languages]
          def check(name):
              url = base + name
              row = {'path': name, 'url': url}
              try:
                  req = urllib.request.Request(url, headers={'User-Agent':'GPM-Pages-verification/1.0','Cache-Control':'no-cache'})
                  with urllib.request.urlopen(req, timeout=20) as response:
                      body = response.read()
                      row.update(status=response.status, matches_source=body == (root / name).read_bytes(), size=len(body), sha256=hashlib.sha256(body).hexdigest())
              except Exception as error:
                  row.update(status='error', matches_source=False, error=str(error))
              return row
          rows = []
          for attempt in range(12):
              with concurrent.futures.ThreadPoolExecutor(max_workers=6) as pool:
                  rows = list(pool.map(check, names))
              failures = [row for row in rows if row['status'] != 200 or not row['matches_source']]
              print(f'Attempt {attempt + 1}: {len(rows) - len(failures)}/{len(rows)} live files match tested source.', flush=True)
              if not failures:
                  break
              time.sleep(8)
          out = pathlib.Path('qa/gpm-live'); out.mkdir(parents=True, exist_ok=True)
          report = {'base':base, 'commit':os.environ.get('GITHUB_SHA'), 'checked_files':len(rows), 'failed':len(failures), 'files':rows}
          (out / 'http-report.json').write_text(json.dumps(report, indent=2), encoding='utf8')
          for row in rows:
              print(json.dumps(row))
          with open(os.environ['GITHUB_STEP_SUMMARY'],'a') as summary:
              summary.write(f'## Public GPM helper verification\n{len(rows)} files checked; {len(failures)} mismatches.\n')
          if failures:
              raise SystemExit('Live files do not yet match the tested version.')
          print('LIVE_GPM_VERIFIED: all published files return HTTP 200 and match the tested source.')
          PY
      - name: Save live HTTP evidence
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: gpm-live-verification
          path: qa/gpm-live
          retention-days: 7
