name: Review the 3D percent introduction

on:
  push:
    branches: [feat/scroll-3d-percent]
    paths:
      - '.github/workflows/intro3d-review.yml'
      - 'src/**'
      - 'scripts/**'
      - 'tests/**'
      - 'package.json'
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: intro3d-review
  cancel-in-progress: false

jobs:
  review:
    if: github.repository == 'Maksimas-win/bukiskis-parama' && github.ref == 'refs/heads/feat/scroll-3d-percent'
    runs-on: ubuntu-latest
    timeout-minutes: 15
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v6
        with:
          ref: feat/scroll-3d-percent
      - uses: actions/setup-node@v6
        with:
          node-version: '22'
      - name: Build the complete existing site and run all tests
        env:
          SITE_URL: https://maksimas-win.github.io/bukiskis-parama/
        run: |
          npm run build
          npm test
      - name: Install a development-only browser test runner
        run: |
          npm install --prefix /tmp/intro3d-qa --no-audit --no-fund --save-exact playwright@1.56.1
          /tmp/intro3d-qa/node_modules/.bin/playwright install --with-deps chromium
      - name: Test desktop, mobile, six languages and fallbacks
        env:
          PLAYWRIGHT_PACKAGE_JSON: /tmp/intro3d-qa/package.json
        run: node scripts/check-intro3d.mjs
      - name: Save the tested generated site to the isolated branch
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add docs
          if ! git diff --cached --quiet; then
            git commit -m 'Build tested 3D introduction for GitHub Pages in six languages'
            git push origin HEAD:feat/scroll-3d-percent
          fi
      - name: Prepare a review snapshot
        if: always()
        run: |
          mkdir -p qa/intro3d-review
          tar -cJf qa/intro3d-review/source-snapshot.tar.xz src scripts tests docs package.json AGENTS.md README.md
      - name: Upload screenshots and the review snapshot
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: intro3d-review
          path: qa/intro3d-review
          retention-days: 7
