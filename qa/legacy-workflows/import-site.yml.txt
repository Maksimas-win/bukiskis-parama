name: Import informational site v0.2.0

on:
  push:
    branches: [main]
    paths:
      - bukiskis-parama-v0.2.0.zip
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: import-bukiskis-parama
  cancel-in-progress: false

jobs:
  import:
    if: github.repository == 'Maksimas-win/bukiskis-parama' && github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - name: Check out this repository
        uses: actions/checkout@v6
        with:
          ref: main

      - name: Check for the approved archive
        id: archive
        shell: bash
        run: |
          if [ -f bukiskis-parama-v0.2.0.zip ]; then
            echo 'present=true' >> "$GITHUB_OUTPUT"
          else
            echo 'present=false' >> "$GITHUB_OUTPUT"
            echo 'Upload bukiskis-parama-v0.2.0.zip to the root of this repository. The import will then run automatically.' >> "$GITHUB_STEP_SUMMARY"
          fi

      - name: Set up Node.js
        if: steps.archive.outputs.present == 'true'
        uses: actions/setup-node@v6
        with:
          node-version: '22'

      - name: Verify and unpack the approved release
        if: steps.archive.outputs.present == 'true'
        shell: bash
        run: |
          python3 - <<'PY'
          from pathlib import Path, PurePosixPath
          import hashlib
          import zipfile

          root = Path.cwd().resolve()
          archive = root / 'bukiskis-parama-v0.2.0.zip'
          expected = '18aee0e1903127682b639af68fbc323a122d3db2e9c9bbceca778703a687176c'
          actual = hashlib.sha256(archive.read_bytes()).hexdigest()
          if actual != expected:
              raise SystemExit('Archive checksum does not match the approved v0.2.0 release. No files were imported.')

          # This is an initial import, not a replacement for subsequent development.
          # Refuse to overwrite site sources that may have been edited meanwhile.
          if (root / 'src/site.config.json').exists() or (root / 'docs/index.html').exists():
              raise SystemExit('Site files already exist. Initial import was stopped to preserve subsequent changes.')

          with zipfile.ZipFile(archive) as bundle:
              members = bundle.infolist()
              if len(members) > 200 or sum(m.file_size for m in members) > 10_000_000:
                  raise SystemExit('Unexpected archive size.')
              prepared = []
              for member in members:
                  p = PurePosixPath(member.filename)
                  if p.is_absolute() or '..' in p.parts or not p.parts or p.parts[0] != 'bukiskis-parama':
                      raise SystemExit('Unexpected archive path.')
                  if member.is_dir():
                      continue
                  relative = PurePosixPath(*p.parts[1:])
                  if not relative.parts or relative.parts[0] in {'.git', '.github', '.bootstrap'}:
                      raise SystemExit('Disallowed archive member.')
                  if (member.external_attr >> 16) & 0o170000 == 0o120000:
                      raise SystemExit('Symbolic links are not permitted.')
                  target = (root / str(relative)).resolve()
                  if not target.is_relative_to(root):
                      raise SystemExit('Archive path escapes the repository.')
                  prepared.append((target, bundle.read(member)))
              for target, content in prepared:
                  target.parent.mkdir(parents=True, exist_ok=True)
                  target.write_bytes(content)
          print(f'Imported {len(prepared)} files from the approved release.')
          PY

      - name: Rebuild the six-language static site
        if: steps.archive.outputs.present == 'true'
        run: npm run build

      - name: Run the release tests
        if: steps.archive.outputs.present == 'true'
        run: npm test

      - name: Record the import and next publication step
        if: steps.archive.outputs.present == 'true'
        shell: bash
        run: |
          python3 - <<'PY'
          from pathlib import Path
          readme = Path('README.md')
          text = readme.read_text(encoding='utf-8')
          old = '**Статус:** исходники и готовая папка `docs` собраны локально. Новый репозиторий и публикация на GitHub в рамках этой сборки не создавались. Существующие сайты прихода не изменялись.'
          new = '**Статус:** исходники версии 0.2.0 импортированы в этот репозиторий; папка `docs` пересобрана, автоматические тесты пройдены. Доступность сайта через GitHub Pages проверяется отдельно. Существующие сайты прихода не изменялись.\n\n**Первое включение сайта:** Settings → Pages → Deploy from a branch → main → /docs → Save. Дождитесь успешного развёртывания Pages. Сам импорт ещё не означает, что сайт опубликован.'
          if old not in text:
              raise SystemExit('Unexpected README content; review before committing.')
          readme.write_text(text.replace(old, new, 1), encoding='utf-8')
          PY

      - name: Commit the site and sources to main
        if: steps.archive.outputs.present == 'true'
        shell: bash
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add -- .gitattributes .gitignore AGENTS.md CHANGELOG.md README.md SOURCES.md START_HERE.txt docs package.json preview qa scripts src tests
          if git diff --cached --quiet; then
            echo 'No site changes to commit.'
          else
            git commit -m 'Import approved informational site v0.2.0 and build six languages'
            git push origin HEAD:main
          fi
          {
            echo '## The informational site is imported'
            echo ''
            echo 'Version 0.2.0: source files, six language versions, and the built docs directory are now in main.'
            echo 'Build and release tests passed before this commit.'
            echo ''
            echo '**Next:** Settings → Pages → Deploy from a branch → main → /docs → Save.'
            echo 'This workflow imports the files; GitHub Pages performs the actual website deployment after Pages is enabled.'
          } >> "$GITHUB_STEP_SUMMARY"
